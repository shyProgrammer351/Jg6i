문제

캐릭터가 칼(/), 창(->), 방패([), 활(D) 중 아이템 하나를 획득하기 위해 한 칸 이동하는 문제입니다 이를 해결하는 프로그램을 작성하세요 
      / 
  [   옷   D
      -> 
이동키(WASD)를 입력하세요: a
	
      옷]      방패 획득
 


      / 
  [   옷   D
      -> 
이동키(WASD)를 입력하세요: W
	
      옷/      칼 획득
 


      / 
  [   옷   D
      -> 
이동키(WASD)를 입력하세요: s
	
      옷->     창 획득

알고리즘 설계

프로그램 시작 
   캐릭터를 중심으로 상하좌우에 놓인 아이템을 출력 
   캐릭터의 이동 방향(WASD)을 입력받기 
   입력에 따른 이동 결과 출력 
      W는 위로 이동해서 칼을 획득 
      A는 왼쪽으로 이동해서 방패를 획득 
      S는 아래로 이동해서 창을 획득 
      D는 오른쪽으로 이동해서 활을 획득 
프로그램 종료

프로그램 구현

     
  
 ※ 단계별 점검: A키(이전), D키(다음) 
   
	// 파일명 : ./Chapter06/WeaponItem.java

	import java.util.Scanner;

		  

	public class WeaponItem

	{

		// 프로그램 시작 

1		public static void main( String[] args ) {

			Scanner scan = new Scanner( System.in );

			// 캐릭터를 중심으로 상하좌우에 놓인 아이템을 출력 

			System.out.println( "     /    " );

			System.out.println( " [  옷   D" );

2			System.out.println( "     ->   \n" );

		  

			// 캐릭터의 이동 방향(WASD)을 입력받기 

			System.out.print( "이동키(WASD)를 입력하세요: " );

3			char direction= scan.next().charAt(0);

		  

			// 입력에 따른 이동 결과 출력 

4			switch( direction ) {

				// W는 위로 이동해서 칼을 획득  

5				case 'w' :  case 'W' :

6					System.out.print( "\n     옷/      칼 획득 " );

7					break;

				// A는 왼쪽으로 이동해서 방패를 획득 

8				case 'a' :  case 'A' :

9					System.out.print( "\n     옷]     방패 획득" );

10					break;

				// S는 아래로 이동해서 창을 획득 

11				case 's' :  case 'S' :

12					System.out.print( "\n     옷->     창 획득" );

13				break;

				// D는 오른쪽으로 이동해서 활을 획득 

14				case 'd' :  case 'D' :

15					System.out.print( "\n     옷D      활 획득" );

			}

			scan.close();

		// 프로그램 종료 

16		}

	}
